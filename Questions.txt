3)Create calculator with arithmetic operators 
#Get 2 numbers from user
read -p "Enter first number: "a
read -p "Enter second number: "b

#get the operators from the user
echo "Choose Operation : + - * /"
read -p "Enter Operator: "op

#perform the calculations
case $op in
+) result = $((a+b))
	echo $result;;
-) result = $((a-b))
	echo $result;;
\*) result = $((a*b))
	echo $result;;
/) if["$b" -ne 0];then
	result=$(echo "scale = 2; $a/$b")
	echo "Result : $a/$b=$result"
	else
	echo "Error: Division by zero not allowed"
	fi;;
	esac

7) create a simple calculator using expr command for question 3
#Get 2 numbers from user
read -p "Enter first number: "a
read -p "Enter second number: "b

#get the operators from the user
echo "Choose Operation : + - * /"
read -p "Enter Operator: "op

#perform the calculations
case $op in
+) result = $(expr $a+$b)
	echo $result;;
-) result = $(expr $a-$b)
	echo $result;;
\*) result = $(expr $a \* $b)
	echo $result;;
/) if["$b" -ne 0];then
	result=$(echo "scale = 2; $a/$b")
	echo "Result : $a/$b=$result"
	else
	echo "Error: Division by zero not allowed"
	fi;;
	*)
	echo "Invalid operator";;
	esac
