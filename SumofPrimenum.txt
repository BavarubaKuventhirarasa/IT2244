Q3>sum of prime numbers between 1 and 100: 1060
sum=0

for ((num=2; num<=100; num++))
do
    is_prime=1
    for ((i=2; i*i<=num; i++))
    do
        if (( num % i == 0 ))
        then
            is_prime=0
            break
        fi
    done

    if (( is_prime == 1 ))
    then
        sum=$(($sum + $num))
    fi
done

echo "Sum of prime numbers between 1 and 100 is: $sum"
